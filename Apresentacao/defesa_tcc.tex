\section{Problema e objetivo}

\begin{frame}{Objetivo do trabalho}{Predição diária e multi-horizonte}
  \vspace{0.45em}
  \begin{block}{}
    \centering
    \Large
    Desenvolver e avaliar modelos para estimar a contagem diária de interrupções com início registrado no Distrito Federal.
  \end{block}

  \vspace{0.6em}
  \begin{itemize}
    \item \textbf{Alvo:} identificadores únicos de interrupção com início registrado em cada dia.
    \item \textbf{Escopo:} todas as causas com data válida; sem filtragem climática por fator gerador.
    \item \textbf{Comparação:} XGBoost, Bi-LSTM e Bi-GRU em 1, 3, 7 e 14 dias à frente.
  \end{itemize}

  \begin{block}{Delimitação importante}
    O alvo não é DEC, FEC, duração, consumidores afetados nem causa da interrupção.
  \end{block}

  \source{Monografia, Seções 1.2 e 1.3.}
  \note{% 
    Apresentador: Giovanni. Tempo-alvo: 1 minuto.
    Apresentar o objetivo principal e definir o alvo com cuidado: contagem diária de identificadores únicos de interrupção. Reforçar que não há filtragem climática por fator gerador.
    [Sources] Monografia, Seções 1.2 e 1.3.
  }
\end{frame}

\section{Limitações e aplicação}

\begin{frame}{O escopo dos resultados ainda é restrito}{Limitações}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \textbf{\large Dados}
      \begin{itemize}
        \item Uma única estação meteorológica.
        \item Resolução diária e geografia agregada.
        \item Ausência de topologia, idade dos ativos e vegetação.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\large Modelagem}
      \begin{itemize}
        \item Redes sem validação temporal interna.
        \item Resíduos ainda autocorrelacionados.
        \item Previsões pontuais sem intervalo preditivo operacional.
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.8em}
  \begin{block}{Escopo das conclusões}
    Os resultados descrevem o período e o DF avaliados; não demonstram causalidade climática nem garantem o mesmo desempenho em novos dados.
  \end{block}

  \source{Monografia, Seções 5.8, 5.13 e 6.4.}
  \note{% 
    Apresentador: Mateus. Tempo-alvo: 1 minuto e 5 segundos.
    Mostrar domínio crítico do trabalho: não extrapolar causalidade nem generalização nacional. Enfatizar que há incerteza das métricas, mas ainda não intervalo preditivo para cada dia.
    [Sources] Monografia, Limitações e Trabalhos Futuros.
  }
\end{frame}

\section{Conclusão}

\begin{frame}{Os objetivos do trabalho foram atingidos}{Objetivo e evidência}
  \small
  \begin{description}
    \item[Base integrada e alvo] INMET + ANEEL em 3.073 dias brutos; 3.066 observações finais e alvo reconstruído por identificadores únicos.
    \item[Atributos e protocolo] 40 preditores de clima, calendário e memória, com separação cronológica e avaliação fora da amostra.
    \item[Comparação controlada] XGBoost, Bi-LSTM, Bi-GRU e persistência avaliados nas mesmas datas-alvo e métricas.
    \item[Robustez e interpretação] Bootstrap em blocos, contrastes pareados, ablação e análise residual, sem transformar associação climática em causalidade.
    \item[Multi-horizonte] Avaliação direta em 1, 3, 7 e 14 dias; a arquitetura com menor erro observado muda conforme horizonte e métrica.
  \end{description}

  \begin{block}{Síntese}
    Os sete objetivos específicos foram cumpridos; a validação prospectiva continua necessária antes de uso operacional.
  \end{block}

  \source{Monografia, Seções 6.1--6.3.}

\end{frame}

\begin{frame}[noframenumbering]{O alvo foi reconstruído de forma auditável}{Reserva: definição e rastreabilidade}
  \small
  \begin{itemize}
    \item \textbf{Unidade contada:} identificadores únicos de interrupção com início registrado em cada dia.
    \item \textbf{Construção do alvo:} deduplicação por ordem de interrupção e exclusão apenas de registros sem data de início válida.
    \item \textbf{Sem filtragem por causa:} o fator gerador existe na base, mas não foi usado para selecionar ocorrências.
    \item \textbf{Sem interpolação do alvo:} lacunas meteorológicas pontuais podem ser tratadas antes da engenharia de atributos, mas a variável-resposta não é estimada artificialmente.
    \item \textbf{Escala final:} 3.073 dias brutos; 3.066 observações válidas após a criação de atributos que exigem histórico.
  \end{itemize}

  \vspace{0.6em}
  \begin{block}{Por que isso importa?}
    Essa definição evita inflar a contagem por duplicatas e deixa explícito que estamos prevendo volume diário observado, e não duração, DEC, FEC ou consumidores afetados.
  \end{block}

  \source{Monografia, Seções 1.2, 3.4, 3.5 e 3.14.}
